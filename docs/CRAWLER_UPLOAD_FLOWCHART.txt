                       data.sql             git push


  步驟 4
  通知組員

./scripts/export_database.sh

# 2. 提交
git add init/outfit_db_with_data.sql
git commit -m "更新資料庫:新增 XXX 個商品"
git push origin Crawler&Detection

# 3. 通知組員
echo "📢 資料庫已更新!請 git pull 並重新匯入"


┌──────────────────────────────────────────────────────────────┐
│ 常見錯誤 ❌                                                   │
└──────────────────────────────────────────────────────────────┘

❌ 只 push 程式碼,忘記匯出資料庫
   → 組員下載後沒有你的資料

❌ 只 commit CSV 檔案
   → 組員還需要執行匯入腳本

❌ 沒有通知組員
   → 組員繼續用舊資料

✅ 正確:匯出 → Git → 通知
